    private const val LIQUID_SHADER = """
    uniform float2 iResolution;
    uniform float  iTime;
    float hash(float2 p){ return fract(sin(dot(p, float2(127.1,311.7))) * 43758.5453123); }
    float noise(float2 p){
      float2 i=floor(p), f=fract(p);
      float a=hash(i), b=hash(i+float2(1,0)), c=hash(i+float2(0,1)), d=hash(i+float2(1,1));
      float2 u=f*f*(3.0-2.0*f);
      return mix(a,b,u.x) + (c-a)*u.y*(1.0-u.x) + (d-b)*u.x*u.y;
    }
    half4 main(float2 fragCoord){
      float2 uv = fragCoord / iResolution.xy;
      uv.x *= iResolution.x / iResolution.y;
      float t = iTime * 0.06;
      float n1 = noise(uv*2.3 + t);
      float n2 = noise(uv*4.7 - t*0.8);
      float n3 = noise(uv*1.1 + float2(t*0.4, -t*0.3));
      float h  = (n1*0.55 + n2*0.35 + n3*0.10);
      float3 cBase = float3(0.039, 0.063, 0.071); // #0A0F12
      float3 cCyan = float3(0.000, 1.000, 0.816); // #00FFD0
      float3 cBlue = float3(0.094, 0.643, 1.000); // #18A4FF
      float3 col = mix(cBase, cBlue, smoothstep(0.22,0.82,h));
      col = mix(col, cCyan, smoothstep(0.80,1.00,h)*0.85);
      col += hash(fragCoord*0.75)*0.010;
      return half4(col, 1.0);
    }
    """

    object ThemePrefs {
        private const val PREF_NAME = "idrug_theme"
        private const val KEY = "idrug_theme_enabled"

        fun isIdrugEnabled(ctx: Context): Boolean {
            val prefs = ctx.getSharedPreferences(PREF_NAME, Context.MODE_PRIVATE)
            return prefs.getBoolean(KEY, false)
        }

        fun setIdrugEnabled(ctx: Context, enabled: Boolean) {

            prefs.edit { putBoolean(KEY, enabled) }
        }

    class LiquidGlassView @JvmOverloads constructor(
        context: Context,
        attrs: AttributeSet? = null,
        defStyleAttr: Int = 0
    ) : View(context, attrs, defStyleAttr) {

        private val paint = Paint(Paint.ANTI_ALIAS_FLAG)
        private val runtimeShader: RuntimeShader? = if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.TIRAMISU) {
            RuntimeShader(LIQUID_SHADER)
        } else {
            null
        }
        private val fallbackShader: Shader = LinearGradient(
            0f,
            0f,
            0f,
            1200f,
            intArrayOf(0xFF0A0F12.toInt(), 0xFF18A4FF.toInt(), 0xFF00FFD0.toInt()),
            floatArrayOf(0f, 0.65f, 1f),
            Shader.TileMode.CLAMP
        )
        private var startTime = SystemClock.elapsedRealtime()

        init {
            setWillNotDraw(false)
        }

        override fun onAttachedToWindow() {
            super.onAttachedToWindow()
            startTime = SystemClock.elapsedRealtime()
            invalidate()
        }

        override fun onMeasure(widthMeasureSpec: Int, heightMeasureSpec: Int) {
            super.onMeasure(widthMeasureSpec, heightMeasureSpec)
            if (runtimeShader == null) paint.shader = fallbackShader
        }

        override fun onDraw(canvas: Canvas) {
            val w = width.toFloat()
            val h = height.toFloat()
            val shader = runtimeShader
            if (shader != null) {
                val time = (SystemClock.elapsedRealtime() - startTime) / 1000f
                shader.setFloatUniform("iResolution", w, h)
                shader.setFloatUniform("iTime", time)
                paint.shader = shader
                canvas.drawRect(0f, 0f, w, h, paint)
                postInvalidateOnAnimation()
            } else {
                paint.shader = fallbackShader
                canvas.drawRect(0f, 0f, w, h, paint)
            }
        }
    }

    fun ViewGroup.ensureLiquidBackground() {
        if (children.any { it is LiquidGlassView }) return
        val background = LiquidGlassView(context)
        addView(
            background,
            0,
            ViewGroup.LayoutParams(ViewGroup.LayoutParams.MATCH_PARENT, ViewGroup.LayoutParams.MATCH_PARENT)
        )
    }

    open class ThemeAwareActivity : AppCompatActivity() {
        private var themedRoot: FrameLayout? = null

        override fun setContentView(layoutResID: Int) {
            if (applyIdgThemeIfNeeded { layoutInflater.inflate(layoutResID, null, false) }) return
            super.setContentView(layoutResID)
        }

        override fun setContentView(view: View?) {
            if (view != null && applyIdgThemeIfNeeded { view }) return
            super.setContentView(view)
        }

        override fun setContentView(view: View?, params: ViewGroup.LayoutParams?) {
            if (view != null && applyIdgThemeIfNeeded({ view }, params)) return
            super.setContentView(view, params)
        }

        private fun applyIdgThemeIfNeeded(viewFactory: () -> View, params: ViewGroup.LayoutParams? = null): Boolean {
            if (!ThemePrefs.isIdrugEnabled(this)) return false
            val root = themedRoot ?: ThemeContainer(this).also { themedRoot = it }
            root.removeAllViews()
            root.ensureLiquidBackground()
            val content = viewFactory()
            root.addView(
                content,
                params ?: FrameLayout.LayoutParams(
                    ViewGroup.LayoutParams.MATCH_PARENT,
                    ViewGroup.LayoutParams.MATCH_PARENT
                )
            )
            super.setContentView(root)
            return true
        }

        private class ThemeContainer(context: android.content.Context) : FrameLayout(context) {
            init {

      private const val LIQUID_SHADER = """
      uniform float2 iResolution;
      uniform float  iTime;
      float hash(float2 p){ return fract(sin(dot(p, float2(127.1,311.7))) * 43758.5453123); }
      float noise(float2 p){
        float2 i=floor(p), f=fract(p);
        float a=hash(i), b=hash(i+float2(1,0)), c=hash(i+float2(0,1)), d=hash(i+float2(1,1));
        float2 u=f*f*(3.0-2.0*f);
        return mix(a,b,u.x) + (c-a)*u.y*(1.0-u.x) + (d-b)*u.x*u.y;
      }
      half4 main(float2 fragCoord){
        float2 uv = fragCoord / iResolution.xy;
        uv.x *= iResolution.x / iResolution.y;
        float t = iTime * 0.06;
        float n1 = noise(uv*2.3 + t);
        float n2 = noise(uv*4.7 - t*0.8);
        float n3 = noise(uv*1.1 + float2(t*0.4, -t*0.3));
        float h  = (n1*0.55 + n2*0.35 + n3*0.10);
        float3 cBase = float3(0.039, 0.063, 0.071); // #0A0F12
        float3 cCyan = float3(0.000, 1.000, 0.816); // #00FFD0
        float3 cBlue = float3(0.094, 0.643, 1.000); // #18A4FF
        float3 col = mix(cBase, cBlue, smoothstep(0.22,0.82,h));
        col = mix(col, cCyan, smoothstep(0.80,1.00,h)*0.85);
        col += hash(fragCoord*0.75)*0.010;
        return half4(col, 1.0);
      }
      """

      class LiquidGlassView @JvmOverloads constructor(
          context: Context,
          attrs: AttributeSet? = null,
          defStyleAttr: Int = 0
      ) : View(context, attrs, defStyleAttr) {

          private val paint = Paint(Paint.ANTI_ALIAS_FLAG)
          private val runtimeShader: RuntimeShader? = if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.TIRAMISU) {
              RuntimeShader(LIQUID_SHADER)
          } else {
              null
          }
          private val fallbackShader: Shader = LinearGradient(
              0f,
              0f,
              0f,
              1200f,
              intArrayOf(0xFF0A0F12.toInt(), 0xFF18A4FF.toInt(), 0xFF00FFD0.toInt()),
              floatArrayOf(0f, 0.65f, 1f),
              Shader.TileMode.CLAMP
          )
          private var startTime = SystemClock.elapsedRealtime()

          init {
              setWillNotDraw(false)
          }

          override fun onAttachedToWindow() {
              super.onAttachedToWindow()
              startTime = SystemClock.elapsedRealtime()
              invalidate()
          }

          override fun onMeasure(widthMeasureSpec: Int, heightMeasureSpec: Int) {
              super.onMeasure(widthMeasureSpec, heightMeasureSpec)
              if (runtimeShader == null) paint.shader = fallbackShader
          }

          override fun onDraw(canvas: Canvas) {
              val w = width.toFloat()
              val h = height.toFloat()
              val shader = runtimeShader
              if (shader != null) {
                  val time = (SystemClock.elapsedRealtime() - startTime) / 1000f
                  shader.setFloatUniform("iResolution", w, h)
                  shader.setFloatUniform("iTime", time)
                  paint.shader = shader
                  canvas.drawRect(0f, 0f, w, h, paint)
                  postInvalidateOnAnimation()
              } else {
                  paint.shader = fallbackShader
                  canvas.drawRect(0f, 0f, w, h, paint)
              }
          }
      }

              if (view != null && applyIdgThemeIfNeeded { view }) return
              super.setContentView(view)
          }

          override fun setContentView(view: View?, params: ViewGroup.LayoutParams?) {
              if (view != null && applyIdgThemeIfNeeded({ view }, params)) return
              super.setContentView(view, params)
          }

          private fun applyIdgThemeIfNeeded(viewFactory: () -> View, params: ViewGroup.LayoutParams? = null): Boolean {
              if (!ThemePrefs.isIdrugEnabled(this)) return false
              val root = themedRoot ?: ThemeContainer(this).also { themedRoot = it }
              root.removeAllViews()
              root.ensureLiquidBackground()
              val content = viewFactory()
              root.addView(
                  content,
                  params ?: FrameLayout.LayoutParams(
                      ViewGroup.LayoutParams.MATCH_PARENT,
                      ViewGroup.LayoutParams.MATCH_PARENT
                  )
              )
              super.setContentView(root)
              return true
          }

          private class ThemeContainer(context: android.content.Context) : FrameLayout(context) {
              init {
                  setBackgroundColor(0xE60A0F12.toInt())
              }
          }
